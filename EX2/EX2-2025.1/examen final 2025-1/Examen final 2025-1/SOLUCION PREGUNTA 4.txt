
CREATE OR REPLACE TRIGGER TR_EMPLEADOS_SUELDO
BEFORE INSERT ON AP_EMPLEADOS
FOR EACH ROW
DECLARE
    V_SUMA NUMBER;
    V_CANT NUMBER;
BEGIN
    SELECT COUNT(*),SUM(SALARIO) INTO V_CANT,V_SUMA 
    FROM AP_EMPLEADOS
    WHERE CARGO_ID= :NEW.CARGO_ID
    AND ACTIVO=1;
    
    IF V_CANT>0 THEN
        :NEW.SALARIO:=V_SUMA/V_CANT;
    ELSE
        :NEW.SALARIO:=0;
    END IF;
END;

------------------------------------------------------------------------

CREATE OR REPLACE TRIGGER TR_ELIMINA_AVION
AFTER UPDATE ON AP_AVIONES
FOR EACH ROW
WHEN (new.ESTADO = 'I')
DECLARE
    CURSOR C_VUELOS IS
    SELECT VUELO_ID 
        FROM AP_VUELOS
        WHERE AVION_ID=:OLD.AVION_ID
        AND FECHA_HORA_PARTIDA>SYSDATE;
BEGIN
    FOR R_VUELOS IN C_VUELOS 
    LOOP


        DELETE FROM AP_BOLETOS
        WHERE VUELO_ID=R_VUELOS.VUELO_ID;

        DELETE FROM AP_EMPLEADOS_VUELOS
        WHERE  VUELO_ID=R_VUELOS.VUELO_ID;

        DELETE FROM AP_VUELOS
        WHERE VUELO_ID=R_VUELOS.VUELO_ID;

    END LOOP;

END;
